```typescript
// Add these utility functions for the Cost Allocation tab

// Calculate total entity allocation percentage across all inputs
const calculateTotalEntityAllocation = (budgetItem: any, entityId: number) => {
  if (!budgetItem || !budgetItem.allocationInputs || budgetItem.allocationInputs.length === 0) {
    return 0;
  }
  
  let totalValue = 0;
  let totalInputValue = 0;
  
  // Sum up allocation across all inputs
  budgetItem.allocationInputs.forEach((input: any) => {
    if (input.ratioValues && input.value) {
      const ratioValues = input.ratioValues;
      const entityRatio = ratioValues.find((rv: any) => rv.entityId === entityId);
      const ratioValue = entityRatio?.value || 0;
      const totalRatioValue = ratioValues.reduce((sum: number, rv: any) => sum + (rv.value || 0), 0);
      
      if (totalRatioValue > 0) {
        const percentage = (ratioValue / totalRatioValue) * 100;
        const allocatedValue = (percentage * input.value) / 100;
        
        totalValue += allocatedValue;
        totalInputValue += parseFloat(input.value);
      }
    }
  });
  
  // Calculate percentage of total
  return totalInputValue > 0 ? (totalValue / totalInputValue) * 100 : 0;
};

// Calculate total allocation percentage (should be ~100%)
const calculateTotalAllocation = (budgetItem: any) => {
  if (!budgetItem || !entities || entities.length === 0) {
    return 0;
  }
  
  return entities.reduce((sum, entity) => {
    return sum + calculateTotalEntityAllocation(budgetItem, entity.id);
  }, 0);
};
```
